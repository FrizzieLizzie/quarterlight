# Track the pages read this weekfrom dataclasses import dataclass @dataclassclass Book:    title: str    pages: int = 0 GOAL_PAGES = 350shelf = [Book("Jane Eyre", 532), Book("Dracula", 418)] def weekly_total(books: list[Book]) -> int:    """Add up the pages on the shelf."""    return sum(b.pages for b in books if b.pages > 0) if weekly_total(shelf) >= GOAL_PAGES:    print("Goal met", True)
